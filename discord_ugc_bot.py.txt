import os
import threading
import discord
from discord.ext import commands
import psycopg2
import math
import re
from flask import Flask
from collections import Counter, defaultdict

TOKEN = os.getenv("TOKEN")
DATABASE_URL = os.getenv("DATABASE_URL")

intents = discord.Intents.default()
intents.message_content = True
bot = commands.Bot(command_prefix="!", intents=intents)

STOP_WORDS = {
    "the","a","an","and","of","in","to","for","is","on","that","by","with",
    "from","as","it","at","be","or","no","not","but","all","are","was","were",
    "they","them","his","her","my","your","its","i","you","he","she","we","me",
    "us","our","new","one","if","so","up","out","just","can","also","do","get",
    "has","had","have","did","more","some","like","this","will","use","used"
}

def get_db():
    return psycopg2.connect(DATABASE_URL, sslmode='require')

def extract_words(text):
    """Clean and split text into meaningful words."""
    if not text:
        return []
    words = re.findall(r"[a-zA-Z]+", text.lower())
    return [w for w in words if len(w) > 2 and w not in STOP_WORDS]

@bot.event
async def on_ready():
    print(f"✅ {bot.user} is online (Advanced Analytics)")

# ---------- !scan_status ----------
@bot.command(name="scan_status")
async def scan_status(ctx):
    conn = get_db(); cur = conn.cursor()
    cur.execute("SELECT COUNT(*) FROM discovered_items")
    discovered = cur.fetchone()[0]
    cur.execute("SELECT COUNT(*) FROM items")
    enriched = cur.fetchone()[0]
    cur.execute("SELECT COUNT(*) FROM items WHERE description IS NOT NULL AND description != ''")
    with_desc = cur.fetchone()[0]
    cur.close(); conn.close()

    embed = discord.Embed(title="📡 Catalog Scan Status", color=0x00aaff)
    embed.add_field(name="🔍 IDs Discovered", value=f"**{discovered:,}**", inline=True)
    embed.add_field(name="📦 Items Analyzed", value=f"**{enriched:,}**", inline=True)
    embed.add_field(name="📝 With Descriptions", value=f"**{with_desc:,}**", inline=True)
    embed.set_footer(text="Run scanner and enricher jobs on Render to grow these numbers.")
    await ctx.send(embed=embed)

# ---------- !trends ----------
@bot.command(name="trends")
async def trends(ctx):
    conn = get_db(); cur = conn.cursor()
    cur.execute("""
        SELECT name, description FROM items 
        WHERE total_sales > 0 
        ORDER BY total_sales DESC 
        LIMIT 1000
    """)
    rows = cur.fetchall()
    cur.close(); conn.close()

    if not rows:
        await ctx.send("No sales data yet. Run the enricher.")
        return

    word_counter = Counter()
    for name, desc in rows:
        word_counter.update(extract_words(name))
        word_counter.update(extract_words(desc))

    embed = discord.Embed(
        title="📈 Top Trend Words",
        description="Based on top 1,000 best-selling items (titles + descriptions).",
        color=0xffaa00
    )
    for i, (word, count) in enumerate(word_counter.most_common(15), 1):
        embed.add_field(name=f"{i}. {word}", value=f"Appears **{count}** times", inline=False)
    await ctx.send(embed=embed)

# ---------- !analyze <keyword> ----------
@bot.command(name="analyze")
async def analyze(ctx, *, keyword: str):
    keyword = keyword.strip().lower()
    if not keyword:
        await ctx.send("❌ Provide a keyword like `!analyze bear`")
        return

    conn = get_db(); cur = conn.cursor()
    cur.execute("""
        SELECT name, description, favorite_count, total_sales FROM items 
        WHERE (LOWER(name) LIKE %s OR LOWER(description) LIKE %s)
          AND (favorite_count > 0 OR total_sales > 0)
        LIMIT 2000
    """, (f"%{keyword}%", f"%{keyword}%"))
    rows = cur.fetchall()

    if not rows:
        await ctx.send(f"No data for `{keyword}` yet. Run the scanner and enricher.")
        cur.close(); conn.close(); return

    word_stats = defaultdict(lambda: {"total_favs": 0, "total_sales": 0, "count": 0})
    for name, desc, favs, sales in rows:
        combined = f"{name} {desc}"
        words = extract_words(combined)
        for w in set(words):  # unique per item
            if w == keyword:
                continue
            word_stats[w]["total_favs"] += (favs or 0)
            word_stats[w]["total_sales"] += (sales or 0)
            word_stats[w]["count"] += 1

    results = []
    for word, stats in word_stats.items():
        if stats["count"] < 3:
            continue
        avg_favs = stats["total_favs"] / stats["count"]
        avg_sales = stats["total_sales"] / stats["count"]
        cur.execute(
            "SELECT COUNT(*) FROM items WHERE LOWER(name) LIKE %s OR LOWER(description) LIKE %s",
            (f"%{word}%", f"%{word}%")
        )
        competition = cur.fetchone()[0]
        demand = (avg_favs * 0.7) + (avg_sales * 10)
        if competition == 0:
            competition = 1
        score = demand / math.log1p(competition)
        results.append((word, avg_favs, avg_sales, competition, score))

    cur.close(); conn.close()
    results.sort(key=lambda x: x[4], reverse=True)

    embed = discord.Embed(
        title=f"🧠 Deep Analysis for `{keyword}`",
        description=f"Analyzed **{len(rows)}** items (titles + descriptions).",
        color=0x00ff88
    )
    for i, (word, avg_f, avg_s, comp, score) in enumerate(results[:10], 1):
        embed.add_field(
            name=f"{i}. {word}",
            value=f"Score: **{score:,.0f}** | Avg Favs: **{avg_f:,.0f}** | "
                  f"Avg Sales: **{avg_s:,.1f}** | Comp: **{comp:,}**",
            inline=False
        )
    embed.set_footer(text="Score = (AvgFavs×0.7 + AvgSales×10) / log(Competition)")
    await ctx.send(embed=embed)

# ---------- !desc_analyze <keyword> ----------
@bot.command(name="desc_analyze")
async def desc_analyze(ctx, *, keyword: str):
    """Analyze ONLY descriptions to find hidden keyword goldmines."""
    keyword = keyword.strip().lower()
    if not keyword:
        await ctx.send("❌ Provide a keyword like `!desc_analyze bear`")
        return

    conn = get_db(); cur = conn.cursor()
    cur.execute("""
        SELECT description, favorite_count, total_sales FROM items 
        WHERE LOWER(name) LIKE %s
          AND description IS NOT NULL AND description != ''
        LIMIT 2000
    """, (f"%{keyword}%",))
    rows = cur.fetchall()

    if not rows:
        await ctx.send(f"No descriptions for `{keyword}` yet.")
        cur.close(); conn.close(); return

    word_counter = Counter()
    for desc, favs, sales in rows:
        word_counter.update(extract_words(desc))

    cur.close(); conn.close()

    embed = discord.Embed(
        title=f"📝 Hidden Keywords in `{keyword}` Descriptions",
        description=f"Analyzed {len(rows)} item descriptions. These are words creators "
                    f"bury in descriptions to rank higher.",
        color=0xaa66ff
    )
    for i, (word, count) in enumerate(word_counter.most_common(15), 1):
        embed.add_field(name=f"{i}. {word}", value=f"Used in **{count}** descriptions", inline=False)
    embed.set_footer(text="Add these to YOUR descriptions for extra search visibility.")
    await ctx.send(embed=embed)
